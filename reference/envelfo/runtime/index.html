def slew_value(
    body: envelfo.Slew, value: float, previous: float | None, seconds: Fraction
) -> float:
    """Limit one observation in signal units per second, without overshoot."""
    if previous is None:
        return value if body.initial is None else body.initial
    elapsed = float(seconds)
    if value >= previous:
        return min(value, previous + body.rise * elapsed)
    return max(value, previous - body.fall * elapsed)

def threshold_value(
    body: envelfo.Threshold, value: float, previous: float | None
) -> tuple[float, str | None]:
    """Observe one grid sample; initialization is silent and deadband holds state."""
    output = float(value >= body.upper) if previous is None else previous
    if previous == 0 and value >= body.upper:
        return 1.0, 'rising'
    if previous == 1 and value <= body.lower:
        return 0.0, 'falling'
    return output, None

def transform_value(
    body: envelfo.Sum | envelfo.Product | envelfo.Affine | envelfo.Quantize,
    signals: dict[str, float],
) -> float:
    if isinstance(body, envelfo.Quantize):
        return envelfo.quantize_value(body, signals[body.input])
    if isinstance(body, envelfo.Affine):
        return body.offset + body.scale * signals[body.input]
    value = 0.0 if isinstance(body, envelfo.Sum) else 1.0
    for name in body.inputs:
        if isinstance(body, envelfo.Sum):
            value += signals[name]
        else:
            value *= signals[name]
    return value

def initial_envelfo(
    definition: envelfo.EnveLFO, at: Fraction, *, seed: int = 0
) -> envelfo.EnvelfoState:
    if isinstance(definition, envelfo.SampleHold):
        random_state, word = random.random_word(seed)
        return envelfo.EnvelfoState(
            runtime=envelfo.SampleHoldState(
                at=at,
                value=definition.minimum
                + (definition.maximum - definition.minimum) * ((word >> 11) / 2**53),
                random_state=random_state,
            )
        )
    if isinstance(definition, envelfo.LFO):
        assert isinstance(definition.rate, Fraction)
        return envelfo.EnvelfoState(
            runtime=envelfo.LFOState(
                at=at,
                position=envelfo.EnvelfoPosition(
                    at=at, coordinate=definition.phase, rate=definition.rate
                ),
            )
        )
    if isinstance(definition, envelfo.Stages):
        stage = next(s for s in definition.stages if s.name == definition.initial_stage)
        initial = (
            stage.envelfo.initial
            if isinstance(stage.envelfo, envelfo.Envelope)
            else 0.0
        )
        assert isinstance(initial, float)
        return envelfo.EnvelfoState(
            runtime=envelfo.StageState(
                at=at,
                stage=stage.name,
                entry_value=initial,
                position=_initial_position(stage.envelfo, at),
            )
        )
    if not isinstance(definition, envelfo.Envelope):
        raise ValueError('envelfo reference must be materialized before use')
    if not isinstance(definition.initial, float):
        raise ValueError('current envelope initial requires a parent stage')
    return envelfo.EnvelfoState(
        runtime=envelfo.EnvelopeState(
            at=at,
            phase='idle' if definition.release or definition.start == 'event' else 'on',
            start_value=definition.initial,
            position=_initial_position(definition, at).model_copy(
                update={
                    'coordinate': Fraction(1)
                    if not any(
                        s.duration is None or s.duration for s in definition.segments
                    )
                    else Fraction(0)
                }
            ),
        )
    )

def envelfo_at(
    definition: envelfo.EnveLFO, state: envelfo.EnvelfoState, at: Fraction
) -> envelfo.EnvelfoValue:
    if isinstance(definition, envelfo.SampleHold) and isinstance(
        state.runtime, envelfo.SampleHoldState
    ):
        if at < state.runtime.at:
            raise ValueError('sample-and-hold observation precedes its state')
        return envelfo.EnvelfoValue(
            value=state.runtime.value, weight=1.0, status='held'
        )
    if isinstance(definition, envelfo.LFO) and isinstance(
        state.runtime, envelfo.LFOState
    ):
        position = state.runtime.position
        if definition.position_driver == envelfo.PositionDriver.transport:
            assert isinstance(definition.rate, Fraction)
            return envelfo.EnvelfoValue(
                value=definition.center
                + definition.depth
                * oscillator.shape_value(
                    definition.shape,
                    (definition.phase + definition.rate * at) % 1,
                    definition.duty_cycle,
                ),
                weight=1.0,
                status='running',
            )
        coordinate = _coordinate_at(position, at)
        age = _age_at(position, at)
        weight = lfo.activation_weight(age, definition.delay, definition.fade_in)
        return envelfo.EnvelfoValue(
            value=definition.center
            + definition.depth
            * oscillator.shape_value(
                definition.shape, coordinate % 1, definition.duty_cycle
            ),
            weight=weight,
            status='running',
        )
    if isinstance(definition, envelfo.Stages) and isinstance(
        state.runtime, envelfo.StageState
    ):
        return advance_envelfo(definition, state, at).value
    if isinstance(definition, envelfo.Envelope) and isinstance(
        state.runtime, envelfo.EnvelopeState
    ):
        return _envelope_value(definition, state.runtime, at)
    raise ValueError('envelfo state does not match its definition')

def envelfo_event(
    definition: envelfo.EnveLFO,
    state: envelfo.EnvelfoState,
    event: envelfo.EnvelfoEvent,
) -> envelfo.EnvelfoState:
    if isinstance(definition, envelfo.SampleHold) and isinstance(
        state.runtime, envelfo.SampleHoldState
    ):
        control.check_order(state.runtime.at, state.runtime.ordinal, event)
        if event.action not in (
            'sample',
            'note_on',
            'note_off',
            'pause',
            'resume',
            'seek',
            'shift',
            'reverse',
        ):
            raise ValueError('sample-and-hold does not accept this event')
        runtime = state.runtime
        if event.action == 'sample':
            random_state, word = random.random_word(runtime.random_state)
            runtime = runtime.model_copy(
                update={
                    'value': definition.minimum
                    + (definition.maximum - definition.minimum)
                    * ((word >> 11) / 2**53),
                    'random_state': random_state,
                }
            )
        return envelfo.EnvelfoState(
            runtime=runtime.model_copy(
                update={'at': event.at, 'ordinal': event.ordinal}
            )
        )
    if definition.position_driver == envelfo.PositionDriver.transport:
        raise ValueError('transport-position LFO does not accept phase commands')
    if isinstance(definition, envelfo.Stages) and isinstance(
        state.runtime, envelfo.StageState
    ):
        return advance_envelfo(definition, state, event.at, event).state
    if isinstance(definition, envelfo.LFO) and isinstance(
        state.runtime, envelfo.LFOState
    ):
        if event.action == 'advance':
            raise ValueError('LFOs do not accept advance events')
        if event.action == 'cue':
            raise ValueError('LFOs do not accept cue events')
        if event.action == 'sample':
            raise ValueError('LFOs do not accept sample events')
        control.check_order(state.runtime.at, state.runtime.ordinal, event)
        position = _position_at(state.runtime.position, event.at)
        if event.action in ('note_on', 'reset', 'transport'):
            if (
                event.action == 'reset'
                or (event.action == 'note_on' and definition.reset == lfo.Reset.trigger)
                or (
                    event.action == 'transport'
                    and definition.reset == lfo.Reset.transport
                )
            ):
                position = position.model_copy(
                    update={'coordinate': definition.phase, 'age': Fraction(0)}
                )
        else:
            position = _position_command(position, event)
        return envelfo.EnvelfoState(
            runtime=envelfo.LFOState(
                at=event.at, ordinal=event.ordinal, position=position
            )
        )
    if isinstance(definition, envelfo.Envelope) and isinstance(
        state.runtime, envelfo.EnvelopeState
    ):
        if event.action in ('cue', 'reset', 'transport', 'rate', 'sample'):
            raise ValueError('envelopes do not accept this event')
        indefinite = _indefinite(definition)
        if indefinite and event.action in ('seek', 'shift'):
            raise ValueError(
                'indefinite envelopes do not accept normalized position events'
            )
        control.check_order(state.runtime.at, state.runtime.ordinal, event)
        runtime = state.runtime
        position = _position_at(runtime.position, event.at)
        phase = runtime.phase
        start_value = runtime.start_value
        traversals = runtime.traversals
        complete_coordinate = runtime.complete_coordinate
        segment_index = runtime.segment_index
        if (
            phase == 'on'
            and complete_coordinate is None
            and definition.repeat_count is not None
        ):
            crossed, boundary = _repeat_crossings(
                definition,
                runtime.position,
                event.at,
                definition.repeat_count - traversals,
            )
            traversals += crossed
            if boundary is not None:
                position = position.model_copy(update={'coordinate': boundary})
                complete_coordinate = _repeat_boundary_coordinate(
                    definition, boundary, runtime.position.direction
                )
        observed = _envelope_value(definition, runtime, event.at)
        if event.action == 'start' or (
            event.action == 'note_on' and definition.start == 'activation'
        ):
            active = observed.status in ('running', 'held')
            if not active or definition.retrigger != envelope.Retrigger.ignore:
                assert isinstance(definition.initial, float)
                start_value = (
                    observed.value
                    if definition.retrigger == envelope.Retrigger.current
                    else definition.initial
                )
                phase = 'on'
                traversals = 0
                complete_coordinate = None
                segment_index = 0
                position = position.model_copy(
                    update={
                        'coordinate': Fraction(0),
                        'age': Fraction(0),
                        'rate': Fraction(1)
                        if indefinite
                        else _envelope_rate(definition.segments),
                    }
                )
        elif event.action == 'note_off':
            if definition.release and phase == 'on' and observed.status != 'complete':
                phase = 'release'
                start_value = observed.value
                segment_index = 0
                position = position.model_copy(
                    update={
                        'coordinate': Fraction(0),
                        'age': Fraction(0),
                        'rate': Fraction(1)
                        if indefinite
                        else _envelope_rate(definition.release),
                    }
                )
        elif event.action == 'advance':
            if phase == 'idle':
                raise ValueError('advance requires an active indefinite segment')
            active_segments = (
                definition.release if phase == 'release' else definition.segments
            )
            segment_index, start_value = _advance_segment(
                active_segments,
                segment_index,
                start_value,
                position.coordinate,
                indefinite,
            )
            position = position.model_copy(update={'coordinate': Fraction(0)})
        else:
            position = _position_command(position, event)
            if event.action == 'shift' and (
                phase == 'release' or definition.playback == envelfo.PlaybackMode.once
            ):
                position = position.model_copy(
                    update={
                        'coordinate': max(
                            Fraction(0), min(Fraction(1), position.coordinate)
                        )
                    }
                )
        return envelfo.EnvelfoState(
            runtime=envelfo.EnvelopeState(
                at=event.at,
                ordinal=event.ordinal,
                phase=phase,
                start_value=start_value,
                position=position,
                traversals=traversals,
                complete_coordinate=complete_coordinate,
                segment_index=segment_index,
            )
        )
    raise ValueError('envelfo state does not match its definition')

def envelope_complete_at(
    definition: envelfo.Envelope, state: envelfo.EnvelfoState
) -> Fraction | None:
    """Release completion in the owning clock, or None while completion is unbounded."""
    runtime = state.runtime
    if not isinstance(runtime, envelfo.EnvelopeState):
        raise ValueError('Envelope completion requires Envelope state')
    if runtime.phase != 'release':
        return None
    parts = definition.release[runtime.segment_index :]
    if any(s.duration is None for s in parts):
        return None
    total = sum(
        (s.duration for s in parts if isinstance(s, segments.Segment)), Fraction(0)
    )
    position = runtime.position
    target = total if _indefinite(definition) else Fraction(1)
    if not total or position.coordinate >= target:
        return position.at
    if position.paused or position.direction == -1 or not position.rate:
        return None
    return position.at + (target - position.coordinate) / position.rate

def advance_envelfo(
    definition: envelfo.EnveLFO,
    state: envelfo.EnvelfoState,
    at: Fraction,
    event: envelfo.EnvelfoEvent | None = None,
) -> envelfo.EnvelfoAdvance:
    """Advance a staged Envelfo, returning its state and ordered output events."""
    body = definition
    runtime = state.runtime
    if not isinstance(body, envelfo.Stages) or not isinstance(
        runtime, envelfo.StageState
    ):
        raise ValueError('advance_envelfo requires staged Envelfo state')
    control.elapsed(at, runtime.at)
    if event is not None:
        if event.at != at:
            raise ValueError('Envelfo event time must equal the advance boundary')
        control.check_order(runtime.at, runtime.ordinal, event)
    current, emitted = _advance_stages(body, runtime, at, event is None)
    if event is not None:
        if event.action in ('pause', 'resume', 'reverse', 'seek', 'shift', 'rate'):
            stage = next(s.envelfo for s in body.stages if s.name == current.stage)
            if (
                event.action in ('seek', 'shift')
                and isinstance(stage, envelfo.Envelope)
                and _indefinite(stage)
            ):
                raise ValueError(
                    'indefinite envelopes do not accept normalized position events'
                )
            position = _position_command(_position_at(current.position, at), event)
            if (
                event.action == 'shift'
                and isinstance(stage, envelfo.Envelope)
                and stage.playback == envelfo.PlaybackMode.once
            ):
                position = position.model_copy(
                    update={
                        'coordinate': max(
                            Fraction(0), min(Fraction(1), position.coordinate)
                        )
                    }
                )
            current = current.model_copy(
                update={
                    'position': position,
                    'cursor_order': -1,
                }
            )
        elif event.action == 'advance':
            stage = next(s.envelfo for s in body.stages if s.name == current.stage)
            if not isinstance(stage, envelfo.Envelope) or current.completed:
                raise ValueError('advance requires an active indefinite segment')
            position = _position_at(current.position, at)
            index, value = _advance_segment(
                stage.segments,
                current.segment_index,
                current.entry_value,
                position.coordinate,
                _indefinite(stage),
            )
            current = current.model_copy(
                update={
                    'at': at,
                    'segment_index': index,
                    'entry_value': value,
                    'stage_done': False,
                    'position': position.model_copy(update={'coordinate': Fraction(0)}),
                    'cursor_order': -1,
                }
            )
            current, output = _advance_stages(body, current, at, True)
            emitted.extend(output)
        else:
            key = f'cue.{event.cue}' if event.action == 'cue' else event.action
            activation = current.activation
            current, output = _stage_transition(body, current, at, key)
            emitted.extend(output)
            if current.activation == activation and not current.completed:
                current, output = _advance_stages(body, current, at, True)
                emitted.extend(output)
        current = current.model_copy(
            update={
                'at': at,
                'ordinal': event.ordinal,
                'cursor_order': current.cursor_order,
            }
        )
    value = _stage_value(body, current, at)
    return envelfo.EnvelfoAdvance(
        state=envelfo.EnvelfoState(runtime=current), value=value, events=emitted
    )

def next_stage_event(
    body: envelfo.Stages,
    state: envelfo.StageState,
    limit: Fraction,
    inclusive: bool = True,
) -> tuple[Fraction, int, str] | None:
    """Find the next natural boundary without advancing state or emitting events."""
    if state.completed:
        return None
    stage = next(s.envelfo for s in body.stages if s.name == state.stage)
    if (
        isinstance(stage, envelfo.Envelope)
        and _indefinite(stage)
        and not state.stage_done
        and all(s.duration == 0 for s in stage.segments[state.segment_index :])
    ):
        return (
            (state.at, 0, 'done')
            if (state.at <= limit if inclusive else state.at < limit)
            else None
        )
    candidates: list[tuple[Fraction, int, str]] = []
    position = state.position
    if position.paused or not position.rate:
        return None
    start = position.coordinate
    forward = position.direction == 1
    if isinstance(stage, envelfo.Envelope):
        if _indefinite(stage):
            remaining = stage.segments[state.segment_index :]
            if any(s.duration is None for s in remaining):
                return None
            target = sum(
                (s.duration for s in remaining if isinstance(s, segments.Segment)),
                Fraction(0),
            )
            if forward and (start < target or start == target and not state.stage_done):
                candidates.append(
                    (state.at + (target - start) / position.rate, 0, 'done')
                )
        elif stage.playback == envelfo.PlaybackMode.once:
            for index, marker in enumerate(stage.markers):
                target = marker.position
                if (target > start if forward else target < start) or (
                    target == start
                    and state.cursor_order >= 0
                    and (
                        index > state.cursor_order
                        if forward
                        else index < state.cursor_order
                    )
                ):
                    time = state.at + abs(target - start) / position.rate
                    candidates.append((time, index, marker.name))
            if forward and (
                start < 1
                or (start == 1 and 0 <= state.cursor_order < len(stage.markers))
            ):
                candidates.append(
                    (state.at + (1 - start) / position.rate, len(stage.markers), 'done')
                )
        elif stage.loop_start is not None:
            candidates.extend(_named_loop_candidates(stage, state))
        else:
            period = 1 if stage.playback == envelfo.PlaybackMode.loop else 2
            for index, marker in enumerate(stage.markers):
                offsets = (
                    [marker.position]
                    if period == 1
                    else [marker.position, 2 - marker.position]
                )
                for offset in offsets:
                    turn = (
                        floor((start - offset) / period) + 1
                        if forward
                        else ceil((start - offset) / period) - 1
                    )
                    target = turn * period + offset
                    candidates.append(
                        (
                            state.at + abs(target - start) / position.rate,
                            index,
                            marker.name,
                        )
                    )
                    if (
                        state.cursor_order >= 0
                        and (start - offset) % period == 0
                        and (
                            index > state.cursor_order
                            if forward
                            else index < state.cursor_order
                        )
                    ):
                        candidates.append((state.at, index, marker.name))
            boundary = floor(start) + 1 if forward else ceil(start) - 1
            order = len(stage.markers)
            candidates.append(
                (
                    state.at + abs(boundary - start) / position.rate,
                    order,
                    'cycle' if period == 1 else 'turned',
                )
            )
            if period == 2 and boundary % 2 == 0:
                candidates.append(
                    (
                        state.at + abs(boundary - start) / position.rate,
                        order + 1,
                        'cycle',
                    )
                )
            if period == 2 and start.denominator == 1 and start % 2 == 0:
                if state.cursor_order == order:
                    candidates.append((state.at, order + 1, 'cycle'))
    else:
        assert isinstance(stage, envelfo.LFO)
        for index, marker in enumerate(stage.markers):
            offset = start - marker.position
            turn = floor(offset) + 1 if forward else ceil(offset) - 1
            target = turn + marker.position
            time = state.at + abs(target - start) / position.rate
            candidates.append((time, index, marker.name))
            if (
                state.cursor_order >= 0
                and offset.denominator == 1
                and (
                    index > state.cursor_order
                    if forward
                    else index < state.cursor_order
                )
            ):
                candidates.append((state.at, index, marker.name))
    available = [
        c for c in candidates if (c[0] <= limit if inclusive else c[0] < limit)
    ]
    if not available:
        return None
    return (
        min(available)
        if forward
        else min(
            available,
            key=lambda c: (
                c[0],
                c[1] if c[2] in ('turned', 'cycle') else -c[1],
            ),
        )
    )
